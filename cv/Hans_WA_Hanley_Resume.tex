\vspace{-1em}
\section{Research Highlights \& Selected Publications}
\vspace{-0.5em}

\subsection*{\href{https://www.usenix.org/system/files/conference/usenixsecurity25/sec25cycle1-prepub-487-hanley.pdf}{Tracking the Takes and Trajectories of News Narratives} \hfill USENIX Security 2025}
\begin{zitemize}
\item Built a narrative-tracking system with LoRA-adapted E5/MPNet encoders and zero-shot stance detection to uncover propaganda and misinformation networks across 4,076 news websites.
\item GPU-accelerated DP-Means with FAISS to cluster 428M passages from 29M articles into 146K narratives.
\end{zitemize}

\vspace{0.25em}
\subsection*{\href{https://aclanthology.org/2025.acl-long.124/}{Hierarchical Level-Wise News Article Clustering via Multilingual Matryoshka Embeddings} \hfill ACL 2025}
\begin{zitemize}
\item Developed multilingual Matryoshka embeddings with controllable semantic granularity and level-wise clustering to recover stories, topics, and themes across 54 languages; achieved Pearson $\rho=0.816$ on extended SemEval 2022 Task 8.
\item Made multilingual news clusters interpretable through English summaries: created 38,830 synthetic multilingual multi-document summarization examples and LoRA-fine-tuned Llama 3.1-8B-Instruct, improving ROUGE-1 from 0.332 to 0.414.
\end{zitemize}

\vspace{0.25em}
\subsection*{\href{https://aclanthology.org/2023.emnlp-main.694/}{TATA: Stance Detection via Topic-Agnostic and Topic-Aware Embeddings} \hfill EMNLP 2023}
\begin{zitemize}
\item Combined topic-aware and topic-agnostic embeddings with triplet/contrastive learning, retrieval-mined pairs, and synthetic data for unseen-topic stance detection; achieved 0.771 zero-shot macro-F1 on VAST.
\end{zitemize}

\vspace{0.25em}
\subsection*{\href{https://www.pnas.org/doi/10.1073/pnas.2420607122}{Across the Firewall: Foreign Media’s Influence on Chinese Social Media} \hfill PNAS 2025}
\begin{zitemize}
\item Built multilingual MPNet/FAISS retrieval and trained three bilingual mDeBERTa paraphrase models to link Weibo narratives to 8.26M articles; used NETINF to quantify Russian media's outsized influence.
\end{zitemize}
